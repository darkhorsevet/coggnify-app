exports.handler = async (event, context) => {
  console.log('Luma function called:', event.httpMethod);
  
  if (event.httpMethod === 'OPTIONS') {
    return {
      statusCode: 200,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'POST, GET, OPTIONS',
        'Access-Control-Allow-Headers': 'Content-Type, Authorization'
      }
    };
  }

  try {
    const body = JSON.parse(event.body || '{}');
    console.log('Request body:', body);
    
    if (body.test) {
      return {
        statusCode: 200,
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          success: true,
          message: 'Function is working correctly!',
          apiKeySet: !!process.env.LUMA_API_KEY,
          timestamp: new Date().toISOString()
        })
      };
    }

    if (!process.env.LUMA_API_KEY) {
      console.error('LUMA_API_KEY environment variable not set');
      throw new Error('API key not configured. Please add LUMA_API_KEY environment variable.');
    }

    console.log('Calling Luma AI API...');

    const lumaResponse = await fetch('https://api.lumalabs.ai/dream-machine/v1/generations', {
      method: 'POST',
      headers: {
        'Authorization': 'Bearer ' + process.env.LUMA_API_KEY,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        prompt: body.prompt || "Create a smooth 360-degree rotation view of this horse for professional veterinary documentation.",
        aspect_ratio: "16:9",
        loop: true
      })
    });

    console.log('Luma API response status:', lumaResponse.status);

    const lumaResult = await lumaResponse.json();
    console.log('Luma API result:', lumaResult);

    return {
      statusCode: lumaResponse.ok ? 200 : lumaResponse.status,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        success: lumaResponse.ok,
        data: lumaResult,
        message: lumaResponse.ok ? 'Successfully started Luma AI generation' : 'Luma AI API error',
        timestamp: new Date().toISOString()
      })
    };

  } catch (error) {
    console.error('Function error:', error);
    
    return {
      statusCode: 500,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        success: false,
        error: error.message,
        message: 'Server error occurred',
        timestamp: new Date().toISOString()
      })
    };
  }
};
